
@"
# Let Git manage line endings for text files, so Windows, Mac and Linux
# do not fight over them. Text files are stored with LF in the repository.
* text=auto

# Source code and documents are always text
*.java text
*.md   text
*.txt  text

# Windows scripts must keep CRLF
*.bat text eol=crlf
*.cmd text eol=crlf

# Binary files are never changed
*.class binary
*.jar   binary
*.png   binary
*.jpg   binary
*.gif   binary
*.pdf   binary
"@ | Set-Content -Encoding ascii .gitattributes
