# Let Git normalize line endings for text files
* text=auto

# Java source and docs are stored with LF
*.java text eol=lf
*.md text eol=lf
*.xml text eol=lf
*.yml text eol=lf

# Windows scripts need CRLF, shell scripts need LF
*.bat text eol=crlf
*.sh text eol=lf

# Never touch binary files
*.png binary
*.jpg binary
*.pdf binary
*.jar binary
*.class binary
